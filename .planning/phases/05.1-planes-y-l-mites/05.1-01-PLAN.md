Table public.clinica_planes (one row per clinic; base plan set by support, trials kept as history columns)
- clinica_id uuid primary key references public.clinicas(id) on delete cascade
- plan text not null default 'gratis' check (plan in ('gratis','pro','clinica','beta'))
- miembros_max int not null default 1 check (miembros_max between 1 and 50)
- fundador boolean not null default false
- pagado_hasta date null (inclusive last paid day, Bogotá)
- prueba_pro_hasta date null; prueba_clinica_hasta date null; prueba_clinica_usada boolean not null default false
- beta_hasta date null (null = Beta without end until support sets it)
- created_at timestamptz not null default now(); updated_at timestamptz not null default now()
- constraint clinica_planes_pagado_requerido check (plan in ('gratis','beta') or pagado_hasta is not null)
- constraint clinica_planes_tramo check (plan <> 'clinica' or miembros_max in (2, 3, 5))
- constraint clinica_planes_prueba_consistente check (prueba_clinica_hasta is null or prueba_clinica_usada)
- RLS: policy clinica_planes_select for select to authenticated using (public.es_veterinario() and clinica_id = public.mi_clinica_id()); NO insert/update/delete policies; revoke all on table from anon; revoke insert, update, delete, truncate on table from authenticated.

Internal function (revoke all from public, anon, authenticated; no grant)

Language sql, stable, security definer, set search_path = public. Candidates (rango, plan, origen, miembros, hasta, gracia): beta (3, 'beta', 'beta', 999, case when fundador then null else beta_hasta end, 0) if plan = 'beta'; pagado (2 for clinica / 1 for pro, plan, 'pagado', miembros_max for clinica / 1 for pro, case when fundador then null else pagado_hasta end, 15) if plan in ('pro','clinica'); prueba_clinica (2, 'clinica', 'prueba_clinica', 5, prueba_clinica_hasta, 15) if prueba_clinica_hasta not null; prueba_pro (1, 'pro', 'prueba_pro', 1, prueba_pro_hasta, 0) if prueba_pro_hasta not null; gratis (0, 'gratis', 'gratis', 1, null, 0) ALWAYS (also when the clinic has no row). Eligible: hasta is null or p_hoy <= hasta + gracia. Chosen: order by rango desc, (origen = 'pagado') desc, (hasta is null or p_hoy <= hasta) desc, hasta desc nulls first, limit 1 — at equal rango a paid plan (vigente or in gracia) wins over a trial (checker warning 10).
